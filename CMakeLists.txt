cmake_minimum_required(VERSION 3.21)
project(Interior3D VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets OpenGL OpenGLWidgets)
add_library(interior_editor src/Scene.cpp src/Viewport.cpp src/TvGreeting.cpp src/Editor.cpp)
target_include_directories(interior_editor PUBLIC src)
target_link_libraries(interior_editor PUBLIC Qt6::Widgets Qt6::OpenGL Qt6::OpenGLWidgets)
if(MSVC)
  target_compile_options(interior_editor PUBLIC /utf-8)
endif()
add_executable(Interior3D src/main.cpp)
target_link_libraries(Interior3D PRIVATE interior_editor)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(interior_tests tests/EditorTest.cpp)
  target_link_libraries(interior_tests PRIVATE interior_editor Qt6::Test)
  add_test(NAME scene COMMAND interior_tests sceneRoundTrip)
  add_test(NAME viewport COMMAND interior_tests viewportWorkflow editorWorkflow projectActions remoteGreeting)
  set_tests_properties(scene PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  set_tests_properties(scene viewport PROPERTIES TIMEOUT 45)
endif()
